cmake_minimum_required(VERSION 4.2)
project(mfs_macos_manager_queries LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(CMAKE_CROSSCOMPILING OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Query consumer requires a native 64-bit host")
endif()
if(WIN32)
    if(NOT CMAKE_GENERATOR STREQUAL "Visual Studio 18 2026")
        message(FATAL_ERROR "Windows portable query model requires VS2026/MSBuild")
    endif()
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
if(NOT EXISTS "${MFS_MEOJSON_INCLUDE}/meojson/json.hpp")
    message(FATAL_ERROR "Explicit prepared meojson required")
endif()
set(OpenCV_STATIC ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../build-support/macos-base-source.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../build-support/macos-manager-source.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../build-support/manager-metadata-source.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../build-support/macos-manager-query-source.cmake")
add_library(mfs_macos_manager_queries SHARED consumer.cpp host.cpp)
target_include_directories(mfs_macos_manager_queries PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/overlay"
    "${CMAKE_CURRENT_SOURCE_DIR}/../macos-screencap/overlay"
    "${CMAKE_CURRENT_SOURCE_DIR}/../macos-screencap"
    "${MFS_UPSTREAM_ROOT}/include" "${MFS_MEOJSON_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(mfs_macos_manager_queries PRIVATE ${OpenCV_LIBS})
set_target_properties(mfs_macos_manager_queries PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/manager-host-scope.txt"
    "Separate host.cpp compiles the exact fixed manager constructor/connect/connected/request_uuid/get_features/screencap/inactive/get_info bodies with explicitly controlled availability, capture and host input-owner types. Input models expose only construction/destruction/features and execute no input actions. Each case compares the same Rust manager's complete two-connect state/events/metadata and output retention. This portable model is not real macOS input construction, screenshot, native OS rejection, arbitrary allocation exhaustion, callback drainage or full P1 acceptance. Existing native Input=None query/capture paths remain separate.\n")
if(APPLE)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
        message(FATAL_ERROR "Current native macOS query gate requires arm64")
    endif()
    enable_language(OBJCXX)
    set_property(TARGET mfs_macos_manager_queries PROPERTY OBJCXX_STANDARD 23)
    set_property(TARGET mfs_macos_manager_queries PROPERTY OBJCXX_STANDARD_REQUIRED ON)
    set_source_files_properties(consumer.cpp PROPERTIES LANGUAGE OBJCXX)
    target_sources(mfs_macos_manager_queries PRIVATE
        "${MFS_UPSTREAM_ROOT}/source/MaaMacOSControlUnit/Screencap/ScreenCaptureKitScreencap.mm")
    target_include_directories(mfs_macos_manager_queries PRIVATE "${MFS_UPSTREAM_ROOT}/source/MaaMacOSControlUnit")
    target_compile_options(mfs_macos_manager_queries PRIVATE -fno-objc-arc -fblocks -Wall -Wextra)
    foreach(_framework AppKit ApplicationServices CoreFoundation CoreMedia CoreVideo ScreenCaptureKit)
        find_library(_mfs_mq_${_framework} NAMES ${_framework} REQUIRED)
        target_link_libraries(mfs_macos_manager_queries PRIVATE "${_mfs_mq_${_framework}}")
    endforeach()
elseif(MSVC)
    target_compile_options(mfs_macos_manager_queries PRIVATE /utf-8 /EHsc /W4)
else()
    target_compile_options(mfs_macos_manager_queries PRIVATE -Wall -Wextra)
endif()
