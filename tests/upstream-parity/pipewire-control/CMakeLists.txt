cmake_minimum_required(VERSION 4.2)
project(mfs_pipewire_control_model LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(WIN32)
    if(NOT CMAKE_GENERATOR STREQUAL "Visual Studio 18 2026" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The Windows model requires VS2026/MSBuild x64")
    endif()
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    add_compile_options(/utf-8 /EHsc)
    set(OpenCV_ARCH x64)
    set(OpenCV_RUNTIME vc18)
endif()
set(OpenCV_STATIC ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
if(NOT IS_DIRECTORY "${MFS_UPSTREAM_ROOT}")
    message(FATAL_ERROR "Explicit fixed upstream source is required")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../build-support/pipewire-control-source.cmake")
add_executable(mfs_pipewire_control_model ../native/pipewire-control-model.cpp)
target_include_directories(mfs_pipewire_control_model PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(mfs_pipewire_control_model PRIVATE ${OpenCV_LIBS})
