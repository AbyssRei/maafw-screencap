cmake_minimum_required(VERSION 4.2)
project(png_codec_probe LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(MFS_ZLIB_ONLY "Probe the required compression library without PNG" OFF)
if(MFS_ZLIB_ONLY)
    find_package(ZLIB REQUIRED)
    add_executable(zlib_codec_probe main.c)
    target_compile_definitions(zlib_codec_probe PRIVATE MFS_ZLIB_ONLY=1)
    target_link_libraries(zlib_codec_probe PRIVATE ZLIB::ZLIB)
else()
    find_package(PNG REQUIRED)
    add_executable(png_codec_probe main.c)
    target_link_libraries(png_codec_probe PRIVATE PNG::PNG)
    if(UNIX AND NOT APPLE)
        target_link_libraries(png_codec_probe PRIVATE m)
    endif()
endif()
