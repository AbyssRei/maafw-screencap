cmake_minimum_required(VERSION 4.2)
project(mfs_manager_metadata_model LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The current metadata model requires the planned 64-bit targets")
endif()
if(WIN32)
    if(NOT CMAKE_GENERATOR STREQUAL "Visual Studio 18 2026")
        message(FATAL_ERROR "The Windows model requires VS2026/MSBuild")
    endif()
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
if(NOT EXISTS "${MFS_MEOJSON_INCLUDE}/meojson/json.hpp")
    message(FATAL_ERROR "Metadata model requires its explicitly prepared latest formal meojson headers")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../build-support/manager-metadata-source.cmake")
add_executable(mfs_manager_metadata_model model.cpp)
target_include_directories(mfs_manager_metadata_model PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${MFS_UPSTREAM_ROOT}/include" "${MFS_MEOJSON_INCLUDE}")
if(MSVC)
    target_compile_options(mfs_manager_metadata_model PRIVATE /utf-8 /EHsc)
endif()
